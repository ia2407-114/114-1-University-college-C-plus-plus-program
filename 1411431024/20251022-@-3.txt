#include  <stdio.h>
int main() {

    int tern;
    int factorial_result = 1;
    double approximate = 1;

    printf("請輸入一個正整數來計算階乘：");
    scanf_s("%d", & tern);

        for (int i = 1; 1 <= tern; i++) {
            factorial_result *= i; 
            approximate += (1.0 / factorial_result);
        }
        printf("%d 個項次計算e的近似值為：%f\n", tern, approximate);
        return 0;
}
