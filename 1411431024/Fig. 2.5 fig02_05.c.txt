#include <stdio.h>

// function main begins program execution
int main(void)
{
int p, r, q, w, x, y, z;

printf("Enter 1 number\n"); // prompt
scanf_s("%d", &p);
printf("Enter 2 number\n"); // prompt
scanf_s("%d", &r);
printf("Enter 3 number\n"); // prompt
scanf_s("%d", &q);
printf("Enter 4 number\n"); // prompt
scanf_s("%d", &w);
printf("Enter 5 number\n"); // prompt
scanf_s("%d", &x);
printf("Enter 6 number\n"); // prompt
scanf_s("%d", &y);



z = p * r % q + w / x - y;


printf("%d * %d mod %d + %d / %d - %d are the %d\n", p,r,q,w,x,y,z);
return 1;
}
