#include <stdio.h>

int main() {
    int grade;
    int pass_count = 0;
    int fail_count = 0;
    int students = 0;
    float score = 0;
    float average = 0;

    printf("請開始輸入學生成績,輸入負數即可停止。\n");

    do {
        printf("請輸入成績：");
        scanf_s("%d", &grade);

        if (grade >= 0) {
            students++;
            score += grade;
        if (grade >= 60) {
            pass_count++; 
            }
            else {
                fail_count++; 
            }
        }
        } while (grade >= 0);

        printf("輸入結束。\n");

        if(students > 0){
           
           average = score / students;

           
           printf("統計結果\n");
           printf("及格人數：%d\n", pass_count);
           printf("不及格人數:%d\n", fail_count);
           printf("平均成績：%.2f\n", average);

           if ((float)pass_count / students >= 0.5){
               printf("Bouns to the Teacher!\n");
            }
}
        else {
            printf("未輸入任何有效成績，無法計算。\n");
        }
        return 0;
}
