pet.cpp(實作檔)
#include <iostream>
#include <string>
#include "pet.h"
using namespace std;


PET::PET(string name, int age)
{
    petName = name;
    petAge = age;
}


PET::~PET()
{
    cout << "Bye Bye my pet " << petName << "!" << endl;
}


void PET::displayMessage() const
{
    cout << "Welcome to see my pet!" << endl;
    cout << "Pet name: " << petName << endl;
    cout << "Pet age: " << petAge << endl;
}

pet.h(介面檔)
#include <iostream>
#include <string>

class PET
{

    string petName;

#include <iostream>
#include <string>
#include "pet.h"
using namespace std;

int main()
{
    string nameOfPet;
