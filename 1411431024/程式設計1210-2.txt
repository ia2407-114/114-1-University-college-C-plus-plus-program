#include <stdio.h>

#define SIZE 20

// 排序副程式（依成績由小到大）
void sortByScore(int id[], int score[]) {
    int i, j;
    int temp;

    for (i = 0; i < SIZE - 1; i++) {
        for (j = i + 1; j < SIZE; j++) {
            if (score[i] > score[j]) {

                // 交換成績
                temp = score[i];
                score[i] = score[j];
                score[j] = temp;

                // 交換學號
                temp = id[i];
                id[i] = id[j];
                id[j] = temp;
            }
        }
    }
}

// 眾數副程式
int mode(int score[]) {
    int i, j;
    int count, maxCount = 0;
    int modeValue = score[0];

    for (i = 0; i < SIZE; i++) {
        count = 0;
        for (j = 0; j < SIZE; j++) {
            if (score[i] == score[j]) {
                count++;
            }
        }
        if (count > maxCount) {
            maxCount = count;
            modeValue = score[i];
        }
    }
    return modeValue;
}

int main() {
    int id[SIZE];
    int score[SIZE];
    int i;
    int sum = 0;
    float avg;

    // 使用 scanf 輸入資料
    for (i = 0; i < SIZE; i++) {
        printf("請輸入第 %d 位學生的 學號 與 計概成績：", i + 1);
        scanf("%d %d", &id[i], &score[i]);
    }

    // 計算總分
    for (i = 0; i < SIZE; i++) {
        sum = sum + score[i];
    }

    // 計算平均成績
    avg = sum / (float)SIZE;

    // 排序
    sortByScore(id, score);

    // 印出排序後結果
    printf("\n依成績由小到大排序結果：\n");
    printf("學號\t計概成績\n");
    for (i = 0; i < SIZE; i++) {
        printf("%d\t%d\n", id[i], score[i]);
    }

    // 印出眾數與平均
    printf("\n計概成績眾數：%d\n", mode(score));
    printf("全班平均成績：%.2f\n", avg);

    return 0;
}
