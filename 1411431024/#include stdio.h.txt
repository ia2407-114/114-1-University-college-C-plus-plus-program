#include <stdio.h>

int main(void) {
	int a, b, c;
	printf("Enter three different integers:\n");
	scanf_s("%d %d %d", &a, &b, &c);
	printf("Sum is %d\n", a + b + c);
	printf("Average is %fd\n", ((float)a + b + c) / 3);
	printf("The Product is %d\n", a * b * c);

	int max, min
		max = a;
	min = a;

	if (b > max)
	   {
		max = b;
	}
	if (b < min)
	   {
		max = b;
	}

	if (c > max)
	   {
		max = c;
	}
	if (c < min)
	   {
		min = c;
	}
	printf("Largest is : %d\n", max);
	printf("Smallest is : %d\n", min);
	return 0;
}
