#include <stdio.h>

#define SIZE 5

struct Student {
    int id;
    int compSci;
    int math;
    float average;
};

int linearSearch(const struct Student array[], int key, size_t size);

int main(void) {
    struct Student data[SIZE] = {
        {101, 85, 90, 87.5},
        {102, 70, 75, 72.5},
        {103, 92, 88, 90.0},
        {104, 60, 65, 62.5},
        {105, 77, 82, 79.5}
    };

    printf("Enter student ID to search: ");
    int searchKey;
    scanf("%d", &searchKey);

    int index = linearSearch(data, searchKey, SIZE);

    if (index != -1) {
        printf("\nFound at index: %d\n", index);
        printf("Computer Science: %d\n", data[index].compSci);
        printf("Math: %d\n", data[index].math);
        printf("Average: %.2f\n", data[index].average);
    } 
    else {
        puts("Value not found");
    }

    return 0;
}

int linearSearch(const struct Student array[], int key, size_t size) {
    for (size_t n = 0; n < size; ++n) {
        if (array[n].id == key) {
            return (int)n;
        }
    }
    return -1;
}
