#include <stdio.h>


int main(void)
{
    unsigned int counter;
    int pass=0;
    int fail=0;
    int grade;
    int sum;
    float average;
   sum = 0;
    counter = 0;

    do
    {
        printf("Enter grade, -1 to end: ");
        scanf_s("%d", &grade);
            if (grade >= 0)
            {
                sum += grade;
                counter++;
                if (grade >= 60)
                    pass++;
                else
                    fail++;
            }
    } while (grade >= 0);
    if (counter > 0) {
        printf("及格人數:%d\n", pass);
        printf("不及格人數:%d\n", fail);
        printf("平均分數:%.2f\n", (float)sum / counter);

        if ((float)pass / counter >= 0.5);
            printf("bonus to the teacher!\n");
    }
    else {
        printf("No grades were entered\n");
    }
    return 0;
}
