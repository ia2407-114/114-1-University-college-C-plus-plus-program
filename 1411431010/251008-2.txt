#include<stdio.h>

int main(void)
{
int a, b;
float bmi;
printf("Enter height(cm) and weight(kg): ");
scanf_s("%d %d", &a, &b);
bmi = b / ((a / 100.0f) * (a / 100.0f));
printf("bmi is %.2f\n", bmi);
if (bmi < 18.5f)
{
puts("underweight");
}
else {
if (bmi >= 18.5f && bmi <= 24.9f)
{
puts("normal");
}
else {
if (bmi >= 25.0f && bmi <= 29.9f)
{
puts("overweight");
}
else {
puts("obese");
}
}
}
return 0;
}
