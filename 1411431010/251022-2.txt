#include <stdio.h>
#include <math.h>  

int main(void)
{
    float principal = 0;
    float rate = 0;
    float total=0;
    int aims, year, amount;

    printf("請輸入本金:\n");
    scanf("%f", &principal);
    printf("請輸入利率:\n");
    scanf("%f", &rate);
    printf("請輸入目標:\n");
    scanf("%d", &aims);
     printf("請輸入每年定期定額存入金額:\n ");
    scanf("%d", &amount);
total=principal;

    for ( year = 1; total <= aims; year++) {


        total = total * (1 + rate) + amount;

        printf("第%d年的本利和為%.2f\n", year, total);

       
    }
     printf("需%d年達成目標", year-1);
    return 0;
}
