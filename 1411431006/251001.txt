int main(void)
{
    // 題目一、利用輸入輸出函數，輸入以上變數，並且印出所有變數與結果
    int  z, p, r, q, w, x, y, sum;
    printf("Enter 1 z\n"); // prompt
    scanf_s("%d", &p);
    printf("Enter 2 z\n"); // prompt
    scanf_s("%d", &r);
    printf("Enter 3 z\n"); // prompt
    scanf_s("%d", &q);
    printf("Enter 4 z\n"); // prompt
    scanf_s("%d", &w);
    printf("Enter 5 z\n"); // prompt
    scanf_s("%d", &x);
    printf("Enter 6 z\n"); // prompt
    scanf_s("%d", &y);

    z = p * r % q + w / x - y;

    printf("%d = %d * %d %% %d + %d / %d - %d \n", z, p, r, q, w, x, y);

    // 題目二、(MAX.70) 輸入一個整數成績，判斷是否及格，及格輸出"恭喜通過"，不及格輸出"再加油"
    int num;
    printf("輸入成績\n");
    scanf_s("%d", &num);
    if (num >= 60) {
        printf("恭喜通過");
    }
    if (num < 60) {
        printf("再加油");
    }

    return 0;
}
