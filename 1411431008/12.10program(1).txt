#include <stdio.h>
#define SIZE 20 

int main(void)
{   int id[SIZE] = { 12, 13, 14, 15, 16, 17, 19, 21, 22, 23, 24, 26, 27, 28, 29, 31, 32, 33, 36, 37 };
    int score[SIZE] = { 66, 67, 66, 70, 73, 74, 75, 75, 62, 32, 44, 46, 27, 12, 80, 63, 66, 90, 11, 66 };

    int temp_id;    
    int temp_score; 

    printf("原始資料學號:\n");
    for (size_t i = 0; i < SIZE; ++i) {
        printf("%2d ", id[i]);   
    }
    printf("\n");
    printf("原始資料成績:\n");
    for (size_t i = 0; i < SIZE; ++i) {
        printf("%2d ", score[i]);
    }
    
    for (unsigned int pass = 1; pass < SIZE; ++pass) {

        for (size_t i = 0; i < SIZE - pass; ++i) {
            
            if (score[i] > score[i + 1]) {

                
                temp_score = score[i];
                score[i] = score[i + 1];
                score[i + 1] = temp_score;

                
                temp_id = id[i];
                id[i] = id[i + 1];
                id[i + 1] = temp_id;
            }
        }
    }

    printf("\n");
    printf("排序後資料 (成績由小到大):\n");
    printf("%4s%6s\n", "學號", "成績");
   
    for (size_t i = 0; i < SIZE; ++i) {
        printf("%4d%6d\n", id[i], score[i]);
    }
  

    return 0;
}
