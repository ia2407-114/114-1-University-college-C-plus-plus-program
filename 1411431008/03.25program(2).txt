#include <iostream>
#include <string>
using namespace std;

class PetDog {
public:
    void setPetData(string name, int age) {
        if (name.length() >= 2 && name.length() <= 5) {
            petName = name;
        }
        else {
            petName = "Unknown";
        }

        if (age >= 0 && age < 15) {
            petAge = age;
        }
        else {
            petAge = -1;
        }
    }

    
    void getPetData() const {
        char isOwner;
        cout << "是否為主人? (是按 Y ): ";
        cin >> isOwner;

        if (isOwner == 'Y' || isOwner == 'y') {
            cout << "成功\n" << endl;
            cout << "寵物姓名: " << petName << endl;
            cout << "寵物年齡:" << petAge << endl;
        }
        else {
            cout << "失敗" << endl;
        }
    }

private:
    string petName;
    int petAge;
};

int main() {
    PetDog myDog;
    string name;
    int age;

    myDog.setPetData("dodo", 3);
    myDog.getPetData();

    cout << "\n輸入寵物姓名: ";
    cin >> name;
    cout << "輸入寵物年齡: ";
    cin >> age;

    myDog.setPetData(name, age);
    cout << "\n寵物資料" << endl;
    myDog.getPetData();

    return 0;
}
