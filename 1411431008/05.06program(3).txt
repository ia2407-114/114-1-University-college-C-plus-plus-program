#ifndef MATRIX_H
#define MATRIX_H

#include <iostream>

class Matrix {
public:
    int row, column;
    float** element;
    
    Matrix(int m, int n);

    Matrix(char name, int m, int n);
    Matrix operator-(Matrix x);
};

#endif



#include "Matrix.h"
#include <iostream>
using namespace std;


Matrix::Matrix(int m, int n) {
    row = m;
    column = n;
    element = new float* [m];
    for (int i = 0; i < m; i++)
        element[i] = new float[n];
}

Matrix::Matrix(char name, int m, int n) {
    row = m;
    column = n;
    element = new float* [m];
    for (int i = 0; i < m; i++)
        element[i] = new float[n];

    cout << "輸入一" << m << 'x' << n << "矩陣" << name << endl;
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cout << name << '[' << i << "][" << j << "]=";
            cin >> element[i][j];
        }
    }
}

Matrix Matrix::operator-(Matrix x) {
    
    Matrix temp(this->row, this->column);
    for (int i = 0; i < this->row; i++) {
        for (int j = 0; j < this->column; j++) {
            
            temp.element[i][j] = this->element[i][j] - x.element[i][j];
        }
    }
    return temp;
}



#include <iostream>
#include "Matrix.h"
using namespace std;

int main() {
    int i, j, row, column;
   
    cout << "輸入A[]及B[]的列數(row):";
    cin >> row;
    cout << "輸入A[]及B[]的行數(column):";
    cin >> column;
    
    Matrix a('A', row, column);
    Matrix b('B', row, column);

    Matrix c(row, column);
    
    c = a - b;

    cout << "__________" << endl;
    cout << "A[] = \n";
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << a.element[i][j] << '\t';
        }
        cout << endl;
    }

    cout << "__________" << endl;
    cout << "B[] = \n";
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << b.element[i][j] << '\t';
        }
        cout << endl;
    }

    cout << "__________" << endl;
    cout << "A[] - B[]= \n";
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << c.element[i][j] << '\t';
        }
        cout << endl;
    }

    return 0;
}
