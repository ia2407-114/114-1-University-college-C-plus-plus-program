#include <iostream>
#include <string>
using namespace std;

class PetDog {
public:
    void setPetData(string name, int age) {
        if (name.length() > 2 && name.length() < 5) {
            petName = name;
        }
        else {
            petName = "Unknown";
            cout << "姓名長度錯誤 " << endl;
        }

        if (age >=0 && age<20) {
            petAge = age;
        }
        else {
            petAge = 0;
            cout << "年齡錯誤" << endl;
        }
    }
    
    void getPetData() const {
        cout << "寵物姓名: " << petName << endl;
        cout << "寵物年齡: " << petAge << endl;
    }

private:
    string petName;
    int petAge;
};

int main() {
    PetDog myDog;
    string name;
    int age;

    cout << "輸入寵物姓名: ";
    cin >> name;
    cout << "輸入寵物年齡: ";
    cin >> age;

    myDog.setPetData(name, age);
    cout << "\n寵物資料結果" << endl;
    myDog.getPetData();

    return 0;
}
