#ifndef FLIGHT_OBJECTS_H
#define FLIGHT_OBJECTS_H
#include <iostream>
#include <string>

using namespace std;

class flight_object {
public:
    string name;          
    string id;                
    int pilot;            
    float kerosene;                 
    string manufacturer; 
 
    void create_flight_object(string obj);
};


class airliner : public virtual flight_object {
protected:
    int passenger;  
    int service_person; 

public:    
    void create_airliner(); 
    void display();
};


class battleplane : public virtual flight_object {
protected:
    string weapon;

public:
    void create_battleplane();
    void display();
};
 
class bombplane : public airliner, public battleplane {
public: 
    void create_bombplane();
    void display();
    void save_to_file(string filename);
};

#endif

void bombplane::save_to_file(string filename) {
    ofstream outfile;
    outfile.open(filename, ios_base::in);

    if (outfile.fail()) {
        cout << filename << " 檔案無法開啟!\n";
        exit(1);
    }

    outfile << "\n顯示轟炸機的資料\n";
    outfile << "名稱:" << name << "\n";
    outfile << "編號:" << id << "\n";
    outfile << "駕駛員人數:" << pilot << "\n";
    outfile << "煤油量(公升):" << kerosene << "\n";
    outfile << "乘客人數:" << passenger << "\n";
    outfile << "武器名稱:" << weapon << "\n";
    outfile << "製造者:" << manufacturer << "\n";

    outfile.close();
    if (outfile.fail()) {
        cout << filename << " 檔案無法關閉!\n";
        exit(1);
    }
    cout << "\n資料已存入 " << filename << "\n";
}
