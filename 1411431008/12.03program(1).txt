#include <stdio.h>

long long factorial(int k);

int main() {
    int n;
    double e = 0.0; 

    printf("請輸入 n 值 : ");
    scanf_s("%d", &n);

    for (int i = 0; i <= n; i++) {
        e += 1.0 / factorial(i);
    }

    printf("當 n = %d 時，計算出的 e 值為: %.10f\n", n, e);

    return 0;
}


long long factorial(int k) {
    if (k <= 1) {
        return 1;
    }
    else {
        return k * factorial(k - 1);
    }
}
