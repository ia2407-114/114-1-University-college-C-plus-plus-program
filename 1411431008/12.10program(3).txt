#include <stdio.h>
#define SIZE 10

void sort_array(int arr[], int n) {
    int i, j, temp;
    
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}


int find_mode(int arr[], int n) {
    int i, j;
    int max_count = 0; 
    int mode = arr[0]; 
    

    for (i = 0; i < n; i++) {
        int current_count = 0; 

        for (j = 0; j < n; j++) {
            if (arr[j] == arr[i]) {
                current_count++;
            }
        }
       
        if (current_count > max_count) {
            max_count = current_count;
            mode = arr[i];
        }
    }
    
    return mode;
}



int main() {
    int student_id[SIZE] = {11, 15, 13, 18, 12, 17, 14, 19, 16, 20};
    int comp_sci_scores[SIZE] = {85, 92, 78, 85, 90, 85, 76, 95, 88, 85};
    int math_scores[SIZE] = {70, 80, 75, 90, 85, 70, 82, 92, 88, 70};
    
    float average_scores[SIZE];
    
    int i;
   
    printf("學號\t計概成績\t數學成績\n");
    for (i = 0; i < SIZE; i++) {
        printf("%d\t    %d\t\t    %d\t\t\t\n", student_id[i], comp_sci_scores[i], math_scores[i]);
    }
    
    
    for (i = 0; i < SIZE; i++) {
        average_scores[i] = (float)(comp_sci_scores[i] + math_scores[i]) / 2.0;
    }
    
    
    printf("學號\t平均成績\n");
    for (i = 0; i < SIZE; i++) {
        printf("%d\t    %.2f\n", student_id[i], average_scores[i]); 
    }
    
    
    sort_array(student_id, SIZE);
    int id_mode = find_mode(student_id, SIZE);
   
    sort_array(comp_sci_scores, SIZE);
    int comp_sci_mode = find_mode(comp_sci_scores, SIZE);
   
    sort_array(math_scores, SIZE);
    int math_mode = find_mode(math_scores, SIZE);
    
    
    printf("排序後的計概成績：\n");
    for (i = 0; i < SIZE; i++) {
        printf("%d ", comp_sci_scores[i]);
    }
    printf("\n");
    
    printf("排序後的數學成績：\n");
    for (i = 0; i < SIZE; i++) {
        printf("%d ", math_scores[i]);
    }
    printf("\n\n");
    
    
    printf("計概成績的眾數 ：%d\n", comp_sci_mode);
    printf("數學成績的眾數 ：%d\n", math_mode);
    
    return 0;
}
