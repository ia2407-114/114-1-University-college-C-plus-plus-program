#include <stdio.h>

int main() {
    
    int fact;//輸入要幾的n//
    int factorial_result = 1;//階層結果//

    
    printf("請輸入一個正整數來計算階乘：");
    scanf_s("%d", &fact);

    
    for (int i = 1; i <= fact; i++) {
        factorial_result *= i; // factorial_result=factorial_result*i//
    }

    printf("%d的階乘結果為%d",fact,factorial_result);

    return 0; 
}
