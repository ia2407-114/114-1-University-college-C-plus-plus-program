#include <stdio.h>

struct student {
    int id;           
    char name[20];    
    int computer, math; 
    double avg;        
};

int main(void) {
    FILE *cfPtr;
    struct student std = {0, "", 0, 0, 0.0};
    int count = 1; 
    
    if ((cfPtr = fopen("students.dat", "rb+")) == NULL) {
        puts("檔案無法開啟");
    } 
    else {
        printf("輸入學號 (輸入 0 結束): ");
        scanf("%d", &std.id);
    
        while (std.id != 0) {
            printf("請輸入 姓名 電腦成績 數學成績: ");
            scanf("%19s %d %d", std.name, &std.computer, &std.math);
            
            std.avg = (std.computer + std.math) / 2.0;
            
            fseek(cfPtr, (std.id - 1) * sizeof(struct student), SEEK_SET);
            
            fwrite(&std, sizeof(struct student), 1, cfPtr);
            
            printf("\n輸入下一筆學號 (輸入 0 結束): ");
            scanf("%d", &std.id);
            count++;
        }
        fclose(cfPtr);
    }
    return 0;
}
