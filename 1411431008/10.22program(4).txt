#include <stdio.h>

int main() {
    int n;
    int i;         
    double pi_approx = 0.0;
    double sign = 1.0;

    //π = 4 - 4/3 + 4/5 - 4/7 + 4/9 - ...
    printf("請輸入要計算的項目總數 n: ");
    scanf_s("%d", &n);


    for (i = 0; i < n; i++) {
        double denominator = 2.0 * i + 1.0;
        pi_approx += sign * (4.0 / denominator);
        sign *= -1.0;
    }

    printf("使用 %d 個項次計算出的π近似值為: %.15f\n", n, pi_approx);

    return 0;
}
