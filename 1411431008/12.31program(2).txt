#include <stdio.h>
#include <string.h>
#define SIZE 10


int binarySearch(const int idArray[], int key, int low, int high);
int linearSearch(const int idArray[], int key, int size);
void copy1(char* s1, const char* s2);
void sort_array_by_id(char name[][20], int stu[][SIZE], float avg[], int n);
void swap_int(int* ptr1, int* ptr2);
void swap_float(float* ptr1, float* ptr2);

int main() {
    const char* stu_name_init[SIZE] = {
        "Danny", "Mary", "Jimmy", "Peter", "Sue",
        "John", "Hearts", "Diamonds", "Clubs", "Spades"
    };

    char stu1_name[SIZE][20];
    
    int stu[3][SIZE] = {
        { 20, 18, 13, 17, 15, 11, 12, 14, 19, 16 }, 
        { 85, 92, 78, 85, 90, 85, 76, 95, 88, 85 },
        { 70, 80, 75, 90, 85, 70, 82, 92, 88, 70 }
    };
    float average_scores[SIZE];

    
    for (int i = 0; i < SIZE; i++) {
        copy1(stu1_name[i], stu_name_init[i]);
        average_scores[i] = (stu[1][i] + stu[2][i]) / 2.0;
    }

   
    sort_array_by_id(stu1_name, stu, average_scores, SIZE);

    printf("學生資料\n");
    printf("%-10s\t%-10s\t%-10s\t%-10s\t%-10s\n", "姓名", "學號", "計概", "數學", "平均");
    for (int i = 0; i < SIZE; i++) {
        printf("%-10s\t%-10d\t%-10d\t%-10d\t%-10.2f\n",
            stu1_name[i], stu[0][i], stu[1][i], stu[2][i], average_scores[i]);
    }

    int searchKey;
    printf("\n輸入要搜尋的學號: ");
    scanf_s("%d", &searchKey);

    
    int index = binarySearch(stu[0], searchKey, 0, SIZE - 1);

    if (index != -1) {
        printf("找到學號 %d 位在陣列索引: %d\n", searchKey, index);
        printf("學生姓名: %s\n", stu1_name[index]);
        printf("計概成績: %d\n", stu[1][index]);
        printf("數學成績: %d\n", stu[2][index]);
        printf("平均成績: %.2f\n", average_scores[index]);
    }
    else {
        printf("\n搜尋失敗，找不到學號 %d\n", searchKey);
    }

    return 0;
}




int binarySearch(const int idArray[], int key, int low, int high) {
    while (low <= high) {
        int middle =(low+high)/ 2; 

        if (key == idArray[middle]) {
            return middle;
        }
        else if (key < idArray[middle]) {
            high = middle - 1;
        }
        else {
            low = middle + 1;
        }
    }
    return -1;
}


int linearSearch(const int idArray[], int key, int size) {
    for (int i = 0; i < size; i++) {
        if (idArray[i] == key) return i;
    }
    return -1;
}


void copy1(char* s1, const char* s2) {
    for (; ((*s1 = *s2) != '\0'); ++s1, ++s2);
}


void swap_int(int* ptr1, int* ptr2) {
    int temp = *ptr1;
    *ptr1 = *ptr2;
    *ptr2 = temp;
}


void swap_float(float* ptr1, float* ptr2) {
    float temp = *ptr1;
    *ptr1 = *ptr2;
    *ptr2 = temp;
}


void sort_array_by_id(char name[][20], int stu[][SIZE], float avg[], int n) {
    char temp_name[20];
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
           
            if (stu[0][j] > stu[0][j + 1]) {

                for (int k = 0; k < 3; k++) {
                    swap_int(&stu[k][j], &stu[k][j + 1]);
                }

                swap_float(&avg[j], &avg[j + 1]);
               
                copy1(temp_name, name[j]);
                copy1(name[j], name[j + 1]);
                copy1(name[j + 1], temp_name);
            }
        }
    }
}
