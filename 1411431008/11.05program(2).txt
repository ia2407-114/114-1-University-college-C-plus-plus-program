#include <stdio.h>

int years_to_goal(double *principal_ptr, double rate, double deposit, double target);


int main() {
    double p, r, aims, deposit;
    int total_years;

    printf("輸入本金:\n");
    scanf_s("%lf", &p);

    printf("輸入年利率:\n");
    scanf_s("%lf", &r);

    printf("輸入存款目標:\n");
    scanf_s("%lf", &aims);

    printf("輸入每年要定期存入的金額:\n");
    scanf_s("%lf", &deposit);

   
    if (p >= aims) {
        printf("初始本金已經達成或超過存款目標了！\n");
        printf("初始本金: %.2f\n", p);
        printf("存款目標: %.2f\n", aims);
    } 
    else {
        total_years = years_to_goal(&p, r, deposit, aims); 
        printf("在第 %d 年，就可以達成或超越存款目標了。\n", total_years);
        printf("最終本利合為: %.2f\n", p); 
    }

    return 0;
}

int years_to_goal(double *principal_ptr, double rate, double deposit, double target) {
    
    int years = 0; 
    while (*principal_ptr < target) {
           *principal_ptr *= (1.0 + rate);
	   *principal_ptr += deposit;      
           years++; 
           printf("第 %d 年底本利合: %.2f\n", years, *principal_ptr);
    }
    return years;
}
