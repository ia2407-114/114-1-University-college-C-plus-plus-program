#ifndef MATRIX_H
#define MATRIX_H

#include <iostream>
#include <cmath> 

class Matrix {
public:
    int row, column;
    float** element; 

    Matrix(int m, int n);
    Matrix operator!();
};

#endif



#include "Matrix.h"

Matrix::Matrix(int m, int n) {
    row = m;
    column = n;
    element = new float* [m];
    for (int i = 0; i < m; i++)
        element[i] = new float[n];
}

Matrix Matrix::operator!() {
    Matrix temp(row, column);
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < column; j++) {
            
            temp.element[i][j] = std::sqrt(this->element[i][j]);
        }
    }
    return temp;
}



#include <iostream>
#include "Matrix.h"
using namespace std;

int main() {
    int i, j, row, column;
   
    cout << "輸入A[]的列數(row):";
    cin >> row;
    cout << "輸入A[]的行數(column):";
    cin >> column;

    Matrix a(row, column), b(row, column);
    cout << "輸入一" << row << 'x' << column << "矩陣A" << endl;
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << "A[" << i << "][" << j << "]=";
            cin >> a.element[i][j];
        }
    }

    b = !a;

    cout << "A[]開平方根前:\n";
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << a.element[i][j] << '\t';
        }
        cout << endl;
    }

    cout << "A[]開平方根後的結果!():\n";
    for (i = 0; i < row; i++) {
        for (j = 0; j < column; j++) {
            cout << b.element[i][j] << '\t';
        }
        cout << endl;
    }

    return 0;
}
