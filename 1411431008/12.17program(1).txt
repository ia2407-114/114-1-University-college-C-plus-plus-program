#include <stdio.h>
#define STUDENTS 10


void sort_data(int data[][STUDENTS], int row_count, float avg[STUDENTS]) {
    int i, j, k;
    float tempAvg;
    int tempVal;


    for (i = 0; i < STUDENTS - 1; i++) {
        for (j = 0; j < STUDENTS - 1 - i; j++) {
            if (avg[j] < avg[j + 1]) {
                     
                tempAvg = avg[j];
                avg[j] = avg[j + 1];
                avg[j + 1] = tempAvg;

                for (k = 0; k < row_count; k++) {
                    tempVal = data[k][j];
                    data[k][j] = data[k][j + 1];
                    data[k][j + 1] = tempVal;
                }
            }
        }
    }
}

int find_mode(int arr[], int n) {
    int i, j;
    int max_count = 0;
    int mode = arr[0];


    for (i = 0; i < n; i++) {
        int current_count = 0;
        for (j = 0; j < n; j++) {
            if (arr[j] == arr[i]) {
                current_count++;
            }
        }
        if (current_count > max_count) {
            max_count = current_count;
            mode = arr[i];
        }
    }
    return mode;
}


int main() {
    int studentData[3][STUDENTS] = {
        {101, 102, 103, 104, 105, 106, 107, 108, 109, 110}, 
        {70,  85,  60,  85,  90,  70,  55,  85,  90,  65},  
        {80,  75,  60,  75,  95,  80,  60,  75,  88,  70}   
    };
    float averages[STUDENTS];
    int i;

    
    for (i = 0; i < STUDENTS; i++) {
        averages[i] = (studentData[1][i] + studentData[2][i]) / 2.0;
    }

    sort_data(studentData, 3, averages);

    
    int modeCS = find_mode(studentData[1], STUDENTS);
    int modeMath = find_mode(studentData[2], STUDENTS);

  
    printf("學生成績表依平均分排序\n");
    printf("%-10s %-10s %-10s %-10s\n", "學號", "計概", "數學", "平均");

    for (i = 0; i < STUDENTS; i++) {
        printf("%-10d %-10d %-10d %-10.1f\n", 
               studentData[0][i], 
               studentData[1][i], 
               studentData[2][i], 
               averages[i]);
    }
    
    printf("計概成績眾數: %d\n", modeCS);
    printf("數學成績眾數: %d\n", modeMath);

    return 0;
}
