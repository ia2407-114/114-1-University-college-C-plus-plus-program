#include <stdio.h>
#include <math.h> 


double series_e(double x) {
    double sum = 0.0;  
    double term = 1.0; 
    int n = 0;

    
    while (fabs(term) > 1e-15) {
        sum += term; 
        n++;         
        term = term * x / n;
    }

    return sum;
}


int main() {
    double x;

    printf("請輸入 x 的值:\n");
    scanf_s("%lf", &x); 

    //無窮級數計算
    double series_result = series_e(x);
    //庫存函數exp計算
    double exp_result = exp(x);
    //誤差
    double error = series_result - exp_result;

    printf("計算 e^%.2f 的結果 \n", x);
    printf("無窮級數(series_e): %.15f\n", series_result);
    printf("庫存函數(exp): %.15f\n", exp_result);
    printf("誤差 : %.15e\n", error);

    return 0;
}
