#include <stdio.h>

double approximate_pi(int n);

//主
int main() {
    int n;         
    double pi_result; 
    
    printf("輸入要計算的項目總數 n: ");
    scanf_s("%d", &n);
    pi_result = approximate_pi(n);
    printf("使用 %d 個項次計算出的π近似值為: %.15f\n", n, pi_result);

    return 0; 
}

//副
double approximate_pi(int n) {
    int i;
    double pi_approx = 0.0; 
    double sign = 1.0;      
    
    for (i = 0; i < n; i++) {
        double denominator = 2.0 * i + 1.0;
        pi_approx += sign * (4.0 / denominator);
        sign *= -1.0;
    }

    return pi_approx;
}
