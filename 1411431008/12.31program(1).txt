#include <stdio.h>
#include <string.h>
#define SIZE 10

int linearSearch(const int array[], int key, int size)
{   for (int n = 0; n < size; ++n) {
        if (array[n] == key) {
            return n;                      
        }
    }
    return -1;                                   
}

void copy1(char* s1, const char* s2) {
    for (; ((*s1 = *s2) != '\0'); ++s1, ++s2);
}

void sort_array_int(int* ptr1, int* ptr2) {
    int temp = *ptr1;
    *ptr1 = *ptr2;
    *ptr2 = temp;
}

void sort_array_float(float* ptr1, float* ptr2) {
    float temp = *ptr1;
    *ptr1 = *ptr2;
    *ptr2 = temp;
}

void sort_array(char name[][20], int stu[][SIZE], float avg[], int n) {
    int i, j;
    char temp_name[20];

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - 1 - i; j++) {
            if (avg[j] < avg[j + 1]) {
                sort_array_float(&avg[j], &avg[j + 1]);

                copy1(temp_name, name[j]);
                copy1(name[j], name[j + 1]);
                copy1(name[j + 1], temp_name);

                for (int k = 0; k < 3; k++) {
                    sort_array_int(&stu[k][j], &stu[k][j + 1]);
                }
            }
        }
    }
}

int main() {
    const char* stu_name_init[SIZE] = {
        "Danny", "Mary", "Jimmy", "Peter", "Sue",
        "John", "Hearts", "Diamonds", "Clubs", "Spades"
    };

    char stu1_name[SIZE][20];
    int stu[3][SIZE] = {
        { 11, 15, 13, 18, 12, 17, 14, 19, 16, 20 },
        { 85, 92, 78, 85, 90, 85, 76, 95, 88, 85 },
        { 70, 80, 75, 90, 85, 70, 82, 92, 88, 70 }
    };
    float average_scores[SIZE];


    for (int i = 0; i < SIZE; i++) {
        copy1(stu1_name[i], stu_name_init[i]);
    }


    for (int i = 0; i < SIZE; i++) {
        average_scores[i] = (stu[1][i] + stu[2][i]) / 2.0;
    }

    printf("排序前結果\n");
    printf("%-10s\t%-10s\t%-10s\t%-10s\t%-10s\n", "姓名", "學號", "計概", "數學", "平均");
    for (int i = 0; i < SIZE; i++) {
        printf("%-10s\t%-10d\t%-10d\t%-10d\t%-10.2f\n",
            stu1_name[i], stu[0][i], stu[1][i], stu[2][i], average_scores[i]);
    }

    sort_array(stu1_name, stu, average_scores, SIZE);

    printf("\n依平均成績排序後 (由高到低)\n");
    printf("%-10s\t%-10s\t%-10s\t%-10s\t%-10s\n", "姓名", "學號", "計概", "數學", "平均");
    for (int i = 0; i < SIZE; i++) {
        printf("%-10s\t%-10d\t%-10d\t%-10d\t%-10.2f\n",
            stu1_name[i], stu[0][i], stu[1][i], stu[2][i], average_scores[i]);
    }

    int linearSearch(const int array[], int key, int size);
    printf("Enter integer search key: ");
    int searchKey; 
    scanf_s("%d", &searchKey);
    int index = linearSearch(stu[0], searchKey, SIZE);
    printf("搜尋的資料內容為 計概成績%d 數學%d 平均%lf", stu[1][index], stu[2][index], average_scores[index]);

    return 0;
}
