#include <iostream>
#include <stdio.h>
#include <stdlib.h>

int main()
{
	int grade;
	int pass = 0;
	int fail = 0;
	int count = 0;
	int sum = 0;

	do {
		printf("輸入學生成績（負數停止）:\n");
		scanf_s("%d", &grade);
		if (grade >= 0) {
			sum = sum + grade;
			count = count + 1;
		}
		if (grade >= 60) {
			printf("及格\n");
			pass = pass + 1;
		}
		else if (grade<60 && grade>0) {
			printf("不及格\n");
			fail = fail + 1;
		}
	} while (grade >= 0);

	printf("pass number is %d\n", pass);
	printf("fail number is %d\n", fail);
	if (count > 0)
		printf("the average grade is %f\n", (float)sum / count);
	else
		printf("未輸入正確成績");


	if (((float)pass /count) >= 0.5)
		printf("Bonus to the Teacher!");

	return 0;
}
