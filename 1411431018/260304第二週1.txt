#include <stdio.h>
           
struct student {                              
   unsigned int id; 
   char name[20]; 
   int computer, math;
   double avg;
};
int main(void) 
{ 
   FILE *sfPtr;

   if ((sfPtr = fopen("students.dat", "rb+")) == NULL) {
      puts("File could not be opened.");
   } 
   else { 
      struct student client = {0, "", 0,0, 0.0};

      printf("%s", "Enter id"
         " (1 to 100, 0 to end input): ");
      scanf("%d", &client.id);

      while (client.id != 0) { 
         printf("%s", "Enter name, computer, math, avg: ");
        
         fscanf(stdin, "%19s%d%d%lf", client.name, 
            &client.computer, &client.math, &client.avg);

         fseek(sfPtr, (client.id - 1) *              
            sizeof(struct student), SEEK_SET);         
     
         fwrite(&client, sizeof(struct student), 1, sfPtr);
         
         printf("%s", "Enter id: ");
         scanf("%u", &client.id);
      } 

      fclose(sfPtr);
   } 
}
