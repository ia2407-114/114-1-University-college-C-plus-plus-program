#include <iostream>
#include <math.h>
int main()
{
	int c; //定期定額
	int aims;
	float r;
	float total,p;
	int year;

	printf("輸入本金:");
	scanf_s("%f", &p);
	printf("輸入年利率:");
	scanf_s("%f", &r);
	printf("輸入定期定額:");
	scanf_s("%d", &c);
	printf("輸入存款目標:");
	scanf_s("%d", &aims);
	total=p;
	for (year = 1;aims>total;year++) {
		total = total * (1 + r) + c;
		printf("第%d年的複利和is %.2f\n",year,total);


	}
	printf("第%d年可達到目標\n",year-1);
//從鍵盤輸入本金(p=10000) 年利率(r=0.05)，每個年底定期定額存入金額，以及存款目標(aims)印出每年底(複利)本利合 直到達成存款目標，讓使用者知道幾年才能達成目標

	return 0;
}
