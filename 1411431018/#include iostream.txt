#include <iostream>
#include <math.h>
int main()
{
	
	float p, r, total;
	int aims, year;
	printf("輸入本金:\n");
	scanf_s("%f", &p);
	printf("輸入年利率:\n");
	scanf_s("%f", &r);
	printf("輸入目標:\n");
	scanf_s("%d", &aims);
	total = p;
	for (year = 1;total < aims;year++) {
		total = total * (1 + r);
		printf("第%d年的本利合是%.2f\n", year, total);
	}

	printf("需要%d年達成目標", year - 1);

	return 0;
}
