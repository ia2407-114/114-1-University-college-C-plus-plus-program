#include <stdio.h>


int main() {

    int i;
    float pi = 4.0;
    int n;
    printf("輸入總項數:");
    scanf_s("%d", &n);

    for (i = 1;n>=i;i++) {
        if (i % 2 == 0)
            pi = pi + 4.0 / (2 * i + 1);
        else if(i%2!=0)
            pi = pi - 4.0 / (2 * i + 1);
           

    }

    printf("近似pi值是%f", pi);


    return 0;

}
