#include <iostream>
#include <math.h>
int main()
{
	int c; //定期定額
	int aims;
	float r;
	float total,p;
	int year;

	printf("輸入本金:");
	scanf_s("%f", &p);
	printf("輸入年利率:");
	scanf_s("%f", &r);
	printf("輸入定期定額:");
	scanf_s("%d", &c);
	printf("輸入存款目標:");
	scanf_s("%d", &aims);
	total=p;
	for (year = 1;aims>total;year++) {
		total = total * (1 + r) + c;
		printf("第%d年的複利和is %.2f\n",year,total);


	}
	printf("第%d年可達到目標\n",year-1);


	return 0;
}
