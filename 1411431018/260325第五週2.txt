#include <iostream>
#include <string> // program uses C++ standard string class
using namespace std;

class Pet
{
public:
    void setPetName(string name)
    {
        if (name.length() > 2 && name.length() < 5)
            PetName = name;
        else
            cout << "error" << endl;
    }
    void setPetAge(int age)
    {
        if (age >=0)
            PetAge = age;
        else
            cout << "error" << endl;

    }

    string getPetName() const
    {
        return PetName;
    }
    int getPetAge() const
    {
        char ANS;
        cout << "Are you a dog owner??(Y/N)" << endl;
        cin >> ANS;
        if (ANS == 'Y' || ANS =='y')
            return PetAge;
        else {
            cout << "Access Denied!!";
            return -1;
        }
    }

    void displayMessage() const
    {
        cout << "Welcome to see my pet" << endl;
        cout << getPetName() << endl;
        cout << getPetAge() << endl;
    }
private:
    string PetName="Yoyo";
    int PetAge=16;
}; 

int main()
{
    string nameOfPet; 
    int ageOfPet;
    Pet myPet;

    // display initial value of courseName
    cout << "Initial Pet name is: " << myPet.getPetName()<< endl;

    cout << "\nPlease enter the Pet name:" << endl;
    getline(cin, nameOfPet);
    myPet.setPetName(nameOfPet);

    cout << "\nPlease enter the Pet age:" << endl;
    cin >> ageOfPet;
    myPet.setPetAge(ageOfPet);

    cout << endl;
    myPet.displayMessage();

    return 0;
}
