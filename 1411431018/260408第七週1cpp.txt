// Fig. 9.5: Time.cpp
// Member-function definitions for class Time.
#include <iostream>
#include <iomanip>
#include <stdexcept>
#include "Time.h" // include definition of class Time from Time.h
using namespace std;

Time::Time(int year, int month, int date)
{
    setTime(year, month, date);
} // end Time constructor

void Time::setTime(int y, int m, int d)
{
    setYear(y);
    setMonth(m);
    setDate(d);
}

void Time::setYear(int y)
{
    if (y >= 2006 && y <= 2026)
        year = y;
    else
        throw invalid_argument("year must be 2006-2026");
}

void Time::setMonth(int m)
{
    if (m >= 1 && m <= 12)
        month = m;
    else
        throw invalid_argument("month must be 1-12");
} 

void Time::setDate(int d)
{
    if (d >= 1 && d <= 31)
        date = d;
    else
        throw invalid_argument("date must be 1-31");
}

unsigned int Time::getYear() const
{
    return year;
}

unsigned int Time::getMonth() const
{
    return month;
} 

unsigned int Time::getDate() const
{
    return date;
}

void Time::printUniversal() const
{
    cout << setfill('0') << setw(4) << getYear() << "/"
        << setw(2) << getMonth() << "/" << setw(2) << getDate();
} // end function printUniversal

// print Time in standard-time format (HH:MM:SS AM or PM)
void Time::printStandard() const
{
    cout << (getYear() - 1911)
        << "/" << setfill('0') << setw(2) << getMonth()
        << "/" << setw(2) << getDate();
} // end function printStandard
