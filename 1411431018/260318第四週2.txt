// Function template maximum test program.
#include <iostream>
using namespace std;

#include "maximum.h" // include definition of function template maximum

int main()
{

    int int1, int2, int3;

    cout << "Input three int values: ";
    cin >> int1 >> int2 >> int3;


    cout << "The intcube value is: "
        << maximum(int1, int2, int3);


    float float1, float2, float3;

    cout << "\n\nInput three float values: ";
    cin >> float1 >> float2 >> float3;


    cout << "The floatcube value is: "
        << maximum(float1, float2, float3);


    double double1, double2, double3;

    cout << "\n\nInput three double: ";
    cin >> double1 >> double2 >> double3;


    cout << "The doublecube value is: "
        << maximum(double1, double2, double3) << endl;
}
