#include <iostream>

// GradeBook class definition
class PET
{
public:
    explicit PET(string name, int age)
    {
        petName = name;
        petAge = age;
    }
    ~PET()
    {
        cout << "My Yoyo,ByeBye,ILOVEYOU." << " " << "My MaoMao is Fat, My DianDian is Cute." << endl;
    }
    // function that sets the course name
    void setPetName(string name)
    {
        if (name.length() <= 5)
            petName = name; // store the course name in the object
    } // end function setCourseName
    void setPetAge(int age)
    {
        if (age >= 0 && age <= 20)
            petAge = age; // store the course name in the object
    } // end function setCourseName

    // function that gets the course name
    string getPetName() const
    {
        return petName; // return the object's courseName
    } // end function getCourseName

    int getPetAge() const
    {
        return petAge; // return the object's courseName
    } // end function getCourseName

    // function that displays a welcome message
    void displayMessage() const
    {
        // this statement calls getCourseName to get the 
        // name of the course this GradeBook represents
        cout << "Welcome to see my pet\n" << getPetName() << ", and it is " << getPetAge() << " years old!"
            << endl;
    } // end function displayMessage
private:
    string petName; // course name for this GradeBook
    int petAge;
}; // end class GradeBook
