#include <iostream>
#include <math.h>
int main()
{
	
	float p, r, total;
	int aims, year;
	printf("輸入本金:\n");
	scanf_s("%f", &p);
	printf("輸入年利率:\n");
	scanf_s("%f", &r);
	printf("輸入目標:\n");
	scanf_s("%d", &aims);
	total = p;
	for (year = 1;total < aims;year++) {
		total = total * (1 + r);
		printf("第%d年的本利合是%.2f\n", year, total);
	}

	printf("需要%d年達成目標", year - 1);

	return 0;

}

//從鍵盤輸入本金(p=10000) 年利率(r=0.05)，以及存款目標(aims)印出每年底(複利)本利合 直到達成存款目標，讓使用者知道幾年才能達成目標 *****注意! 計算次方項時，不得使用庫存函數pow( , )!!!!
