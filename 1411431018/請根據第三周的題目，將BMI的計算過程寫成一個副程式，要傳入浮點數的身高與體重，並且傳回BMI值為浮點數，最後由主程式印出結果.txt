#include <iostream>
#include <stdio.h>
#include <math.h>

float BMI(float height, float weight);

int main(void) {

	float h, w, result;

	printf("輸入身高(公尺):");
	scanf_s("%f", &h);
	printf("輸入體重:");
	scanf_s("%f", &w);

	result = BMI(h, w);
	printf("your BMI is %.2f\n", result);


	return 0;
}

float BMI(float height, float weight)
{
	float ANSWER;
	ANSWER = weight / (height * height);
	return ANSWER;
}
