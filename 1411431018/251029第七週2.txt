#include <iostream>
#include <stdio.h>
#include <math.h>

float BMI(float height, float weight);

int main(void) {

	float h, w, result;

	printf("輸入身高(公尺):");
	scanf_s("%f", &h);
	printf("輸入體重:");
	scanf_s("%f", &w);

	result = BMI(h, w);
	printf("your BMI is %.2f\n", result);

	if (result < 18.5) {
		printf("Underweight\n");
	}
	else if (result >= 18.5 && result <= 24.9){
		printf("Normal\n");
	}
	else if (result >= 25 && result <= 29.9) {
		printf("Overweight\n");
	}
	else {
		printf("Obese\n");
	}

	return 0;
}

float BMI(float height, float weight)
{
	float ANSWER;
	ANSWER = weight / (height * height);
	return ANSWER;
}
