#include <stdio.h>

int main(void)
{
    FILE* sfPtr; // sfPtr = students.txt file pointer   

    // fopen opens file. Exit program if unable to create file 
    if ((sfPtr = fopen("stdents.txt", "w")) == NULL) {
        puts("File could not be opened");
    }
    
    else {
        puts("Enter the id, name, computer, math");
        puts("Enter Ctrl+z to end input."); //EOF=Ctrl+z
        printf("%s", "? "); 

        unsigned int id; 
        char name[30]; 
        int computer, math;
        double avg; 

        scanf("%d%29s%3d%3d", &id, name,&computer,&math);
        
        while (!feof(stdin)) { //stdin=標準輸入 feof=file end of file
        	avg = ((float)computer + math) / 2;
            fprintf(sfPtr, "%d %s %d %d %.2f\n", id, name, computer,math,avg);
            printf("%s", "? ");
            scanf("%d%29s%3d%3d", &id, name, &computer,&math);
        }

        fclose(sfPtr); // fclose closes file   
    }
}
