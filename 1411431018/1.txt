// Overloaded functions.
#include <iostream>
using namespace std;

int volume(int x)
{
	cout << "cubeside" << x << "'s volume is ";
	return x * x * x;
}

double volume(int r, int h)
{
	cout << "roundradius&height" << r << " " << h << "'s volume is ";
	return 3.14159 * r * r * h;
}

int volume(int l, int w, int h)
{
	cout << "reeelong&width&height" << l << " " << w << " "<< h<< "'s volume is ";
	return l * w * h;
}

int main()
{
	int x, r, h, l, w;

	cout << "Enter one integer:";
	cin >> x;
	cout << volume(x) << endl;
	cout << "Enter two integers:";
	cin >> r >> h;
	cout << volume(r,h) << endl;
	cout << "Enter three integers:";
	cin >> l >> w >> h;
	cout << volume(l,w,h) << endl;
}
