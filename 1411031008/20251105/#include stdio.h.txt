#include <stdio.h>

double pi(int);
double ball(double, double);
double ball_area(double, double);
double circle(double, double, double);
double circle_area(double, double, double);

int main(void)
{
	int num;
	double r,h;
	printf("請輸入項目數量:");
	scanf_s("%d", &num);
	printf("請輸入半徑:");
	scanf_s("%1f", &r);
	printf("請輸入高:");
	scanf_s("%1f", &h);

	double pi_val = pi(num);

	printf("pi的近似值為:%.2f\n", pi_val);
	printf("球的體積近似值為:%.2f\n", ball(pi_val,r));
	printf("球的表面積近似值為:%.2f\n", ball_area(pi_val, r));
	printf("圓柱的體積近似值為:%.2f\n", circle(pi_val, r,h));
	printf("圓柱的表面積近似值為:%.2f\n", circle_area(pi_val, r,h));

	return 0;
}


double pi(int n)
{
	int i;
	double pi = 0.0;
	double sign = 1.0;

	for (i = 0;i < n;i++) {
		double denominator = 2.0 * i + 1.0;
		pi += sign * (4.0 / denominator);
		sign *= -1.0;
	}
	return pi;
}

double ball(double pi, double r)
{
	return(4.0 / 3.0 * pi * r * r * r);
}

double ball_area(double pi, double r)
{
	return 4 * pi * r * r;
}

double circle(double pi, double r, double h)
{
	return pi * r * r * h;
}

double circle_area(double pi, double r, double h)
{
	return 2 * pi * r * r + 2 * pi * r * h;
}
