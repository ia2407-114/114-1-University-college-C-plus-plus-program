	printf("\n題目1\n");

		if (grade >= 60) {

		else {

			fail = fail + 1;
		}

		printf("輸入成績，輸入負數便停止:");
		scanf_s("%d", &grade);
	}

	/*do {

	int fact, i=1, result=1;

	scanf_s("%d", &fact);
