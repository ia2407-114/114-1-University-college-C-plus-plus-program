#include <stdio.h>

struct student {
    int id;
    char name[20];
    int computer, math;
    double avg;
};

int main(void){
    
    unsigned int i;
    
    FILE* cfPtr; 
        
    if ((cfPtr = fopen("student.dat", "wb")) == NULL) {
        puts("File could not be opened.");
    }
    else {
        struct student blankstudent = { 0, "", 0, 0, 0.00 };
                         
        for (i = 1; i <= 100; ++i) {
            fwrite(&blankstudent, sizeof(struct student), 1, cfPtr);
        }

        fclose(cfPtr); 
    }
}
