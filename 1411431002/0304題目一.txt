// Fig. 11.11: fig11_11.c
// Writing data randomly to a random-access file
#include <stdio.h>

// clientData structure definition               
struct studentData {
	int id;
    char name[20];
    int computer, math;
    double avg;
};             
 
int main(void) 
{ 
   FILE *cfPtr; // accounts.dat file pointer

   // fopen opens the file; exits if file cannot be opened
   if ((cfPtr = fopen("student.dat", "rb+")) == NULL) {
      puts("File could not be opened.");
   } 
   else { 
      // create clientData with default information
      struct studentData student = {0, "", 0, 0, 0.00};

      // require user to specify account number
      printf("%s", "Enter account number"
         " (1 to 100, 0 to end input): ");
      scanf("%d", &student.id);

      // user enters information, which is copied into file
      while (student.id != 0) { 
         // user enters last name, first name and balance
         printf("%s", "Enter name, computer, math, avg: ");
        
         // set record lastName, firstName and balance value
         fscanf(stdin, "%14s %9d %9d", student.name, 
            &student.computer, &student.math);
        
        student.avg = (double)(student.computer + student.math) / 2.0;

         // seek position in file to user-specified record   
         fseek(cfPtr, (student.id - 1) *           
            sizeof(struct studentData), SEEK_SET);         

         // write user-specified information in file              
         fwrite(&student, sizeof(struct studentData), 1, cfPtr);

         // enable user to input another account number
         printf("%s", "Enter account number: ");
         fscanf(stdin, "%d", &student.id);
      } 

      fclose(cfPtr); // fclose closes the file
   } 
}
