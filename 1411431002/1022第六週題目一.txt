#include <stdio.h>

int main(void) {

	//題目1: 從鍵盤輸入本金(p=10000) 年利率(r=0.05)，以及存款目標(aims)印出每年底(複利)本利和,
	//直到達成存款目標，讓使用者知道幾年才能達成目標
	//*****注意!計算次方項時，不得使用庫存函數pow(, )!!!!

	printf("\n題目1\n\n");

	int p, year=1, aims; //p為本金,year為年,aims為存款目標
	float r, result = 1.0, n = 1.0; //r為年利率,result為複利結果,n為計數器

	printf("請輸入本金p,年利率r,存款目標aims:");
	scanf_s("%d %f %d", &p, &r, &aims);
		
	for (year = 1;result<aims;year++) {	    	
		n = n * (1.00 + r);
		result = p*n;  //複利公式為本金*((1+利率)的n次方)
		printf("第%d年本利和:%.2f\n", year, result);
	}

	printf("第%d年達到存款目標\n", year-1);


	return 0;
}
