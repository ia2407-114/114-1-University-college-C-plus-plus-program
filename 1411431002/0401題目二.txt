實作檔(pet.cpp)：

#include <iostream>
#include <string>
#include"pet.h"

using namespace std;

Pet::Pet(string name, int age) {
    petName = name;
    petAge = age;
}

Pet::~Pet() {
    cout << "\nBye bye " << petName << " ! " << "This is my pet. Is " << petAge << " years old." << endl;
}

void Pet::setPetName(string name) {
    if (name.length() >= 2 && name.length() <= 5) {
        petName = name;
    }
    else {
        cout << "錯誤：姓名長度不符合規範 (須大於等於2且小於等於5)！" << endl;
        petName = "Unknown";
    }
}

void Pet::setPetAge(int age) {
    if (age >= 0 && age <= 20) {
        petAge = age;
    }
    else {
        cout << "錯誤：年齡不符合規範 (須大於等於0且小於等於20)！" << endl;
        petAge = 0;
    }
}

string Pet::getPetName() const {
    return petName;
}

int Pet::getPetAge() const {
    return petAge;
}

void Pet::displayMessage() const {
    cout << "Welcome to see my pet\n" << getPetName() << "," << getPetAge() << "!" << endl;
}


介面檔(pet.h)：

#include <iostream>
#include <string>
using namespace std;

class Pet {

    public:
        Pet(string name, int age);
        ~Pet();
        void setPetName(string name);
        void setPetAge(int age);
        string getPetName() const;
        int getPetAge() const;
        void displayMessage() const;

    private:
        string petName;
        int petAge;
};
