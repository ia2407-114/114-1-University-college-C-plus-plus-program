#include<stdio.h>

//題目2 :建立兩個大小為10的整數陣列，分別存放學生學號以對應的計概成績。寫一個排序副程式，將兩個陣列傳遞至副程式中，依照計概成績由小到大排序，由主程式印出排序後的結果。
//id[]學號 : {12, 13, 14, 15, 16, 17, 19, 21, 22, 23, 24, 26, 27, 28, 29, 31, 32, 33, 36, 37}
//score[]成績 : {66, 67, 66, 70, 73, 74, 75, 75, 62, 32, 44, 46, 27, 12, 80, 63, 66, 90, 11, 66}

void ans(int id[], int score[]);

int main(void)
{
   
	int id[20] = { 12, 13, 14, 15, 16, 17, 19, 21, 22, 23, 24, 26, 27, 28, 29, 31, 32, 33, 36, 37 };
	int score[20] = { 66, 67, 66, 70, 73, 74, 75, 75, 62, 32, 44, 46, 27, 12, 80, 63, 66, 90, 11, 66 };
	int size = 20;

	ans(id, score);

	
	printf("排序結果(由小到大):\n");
	for (int x = 0;x < size;x++) {
		printf("學號:%d  成績:%d \n", id[x], score[x]);
	}
}
void ans(int id[], int score[]) {
	int temp, size = 20;
	for (int i = 1;i <= size;i++) {
		for (int a = 0;a < (size - 1);a++) {
			if (score[a] > score[a + 1]) {

				
				temp = score[a];
				score[a] = score[a + 1];
				score[a + 1] = temp;

				
				temp = id[a];
				id[a] = id[a + 1];
				id[a + 1] = temp;
			}
		}
	}
}
