#include <stdio.h>

int main(void) {
   	//題目 2: 從鍵盤輸入本金(p=10000) 年利率(r=0.05)，每個年底定期定額存入金額,
	//以及存款目標(aims)印出每年底(複利)本利和 直到達成存款目標，讓使用者知道幾年才能達成目標
	//*****注意!計算次方項時，不得使用庫存函數pow(, )!!!!

	printf("\n\n題目2\n\n");

	int p2, year2=1, aims2, save; //p2為本金,year2為年,aims2為存款目標,save為每期存入金額
	float r2, result2 = 1.0, n2 = 1.0; //r2為年利率,result2為複利結果,n2為計數器
	
	printf("請輸入本金,年利率,每年存入金額,存款目標:");
	scanf_s("%d %f %d %d", &p2, &r2, &save, &aims2);

	result2 = p2;

	for (year2 = 1;result2<aims2;year2++) {
	    
	    if (year2==1){	       
	        result2 = result2 * (1.00 + r2);
	    }
	    else{
	        result2 = result2 * (1.00 + r2);
	        result2 = result2 + save ;
	    }	   
		
		printf("第%d年本利和:%.2f\n", year2, result2);
	}

	printf("第%d年達到存款目標\n", year2 - 1);


	return 0;
}
