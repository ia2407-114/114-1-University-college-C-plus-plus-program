int ans (int n );
double ans1(int n1 , double x1);

int main() {
    int n;
    double x , e=1.0 ;
    printf("請輸入n:");
    scanf("%d/n",&n);
    printf("請輸入x:");
    scanf("%f/n",&x);
    for (i=1;i<=n;i++){
        e=e+ans(i)+ans1(i,x);
    }
    printf("e^x=%.5f",e);
    return 0;
}

int ans (int n){
    if (n==0){
        return 1;
    }
    else{
        return n*ans(n-1);
    }
}

double ans1(int n1 , double x1){
    if (n1==0){
        return 1.0;
    }
    else{
        return x1*ans1(x,n-1);
    }
}
