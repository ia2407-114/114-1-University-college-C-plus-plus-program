	//第二題:計算BMI

	float wei, hei, bmi;
	printf("Enter weightInKilograms:\n");
	scanf_s("%f", &wei);
	printf("Enter heightInMeters:\n");
	scanf_s("%f", &hei);

	bmi = wei / (hei * hei);

	printf("BMI is %.2f\n", bmi);
	
	if (bmi < 18.5) {
		printf("Underweight\n");
	}
	else if (18.5 <= bmi and bmi <= 24.9) {
		printf("Normal\n");
	}
	else if (25 <= bmi and bmi <= 29.9) {

	//題目3: 輸入學生成績，直到輸入負數即停止，並且印出學生的平均成績(浮點數)、最高成績、最低成績。

		}

		}

	} 
		
	if (coun != 0) {
		avg = (float)total / coun;
			
		printf("Class average is %.2f\n", avg);
