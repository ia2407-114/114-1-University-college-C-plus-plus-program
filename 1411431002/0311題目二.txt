#include <stdio.h>
#include <stdlib.h>
#include <string.h>
                     
struct listNode {                                      
   char data; 
   struct listNode *nextPtr; 
   int id;
   char name[20];
   int computer, math;
   double avg;
};

typedef struct listNode ListNode; 
typedef ListNode *ListNodePtr; 

void insert(ListNodePtr *sPtr, int id, char name[], int computer, int math);
void printList(ListNodePtr currentPtr);
void delete(ListNodePtr* sPtr, int id);

int main(void)
{ 
   	ListNodePtr startPtr = NULL; 
   	int id, computer, math;
   	char name[20];
   	
	while (1) {
        printf("\n請輸入學號(輸入0結束): ");
        scanf("%d", &id);
        
        if (id == 0) break;

        printf("請輸入姓名: ");
        scanf("%19s", name); 
        printf("請輸入電腦成績: ");
        scanf("%d", &computer);
        printf("請輸入數學成績: ");
        scanf("%d", &math);
        
        insert(&startPtr, id, name, computer, math);

    }
    puts("\n--- 輸入結束，最終結果如下 ---");
    printList(startPtr);
    
    puts("\n--- 刪除學生資料 ---");
    while (startPtr != NULL) { 
        printf("\n請輸入欲刪除的學號(輸入 0 結束): ");
        scanf("%d", &id);

        if (id == 0) break;

        delete(&startPtr, id);
        printList(startPtr);
    }
    
    puts("End of run.");
    return 0;
    
}             



void insert(ListNodePtr *sPtr, int id, char name[], int computer, int math){
 
   ListNodePtr newPtr = malloc(sizeof(ListNode)); 

   if (newPtr != NULL) {
   	newPtr->id = id; 
    strcpy(newPtr->name, name); 
	newPtr->computer = computer;
    newPtr->math = math;
    newPtr->avg = (computer + math) / 2.0;
    newPtr->nextPtr = NULL;

    ListNodePtr previousPtr = NULL;
    ListNodePtr currentPtr = *sPtr;

    
    while (currentPtr != NULL && id > currentPtr->id) {
    	previousPtr = currentPtr;            
        currentPtr = currentPtr->nextPtr; 
    }                                          
    
    if (previousPtr == NULL) { 
	    newPtr->nextPtr = *sPtr;
        *sPtr = newPtr;
    } 
    else { 
        previousPtr->nextPtr = newPtr;
        newPtr->nextPtr = currentPtr;
      } 
   } 
   else {
      printf("%d not inserted. No memory available.\n", id);
   } 
} 

void printList(ListNodePtr currentPtr){ 
   
   if (currentPtr == NULL) {
    puts("List is empty.\n");
   } 
   else {   	
   	printf("\n%-5s %-10s %-5s %-5s %-5s\n", "ID", "Name", "Computer", "Math", "Avg");
    printf("------------------------------------------\n");
   	while (currentPtr != NULL) {
    	printf("%-5d %-10s %-8d %-5d %-5.2f\n",
				currentPtr->id, currentPtr->name,
                currentPtr->computer, currentPtr->math, currentPtr->avg);
            currentPtr = currentPtr->nextPtr;
        }
        printf("------------------------------------------\n");
   } 
} 

void delete(ListNodePtr* sPtr, int id) {
    
    if (*sPtr == NULL) {
        printf("串列為空，無法刪除。\n");
        return;
    }

    ListNodePtr previousPtr = NULL;
    ListNodePtr currentPtr = *sPtr;
    
    while (currentPtr != NULL && currentPtr->id != id) {
        previousPtr = currentPtr;
        currentPtr = currentPtr->nextPtr;
    }

    if (currentPtr != NULL) { 
        ListNodePtr tempPtr = currentPtr; 

        if (previousPtr == NULL) {
            *sPtr = (*sPtr)->nextPtr;
        }
        else {
            previousPtr->nextPtr = currentPtr->nextPtr;
        }

        free(tempPtr);
        printf("學號%d的資料已刪除。\n", id);
    }
    else {
        printf("找不到學號為 %d 的學生資料。\n", id);
    }
}
