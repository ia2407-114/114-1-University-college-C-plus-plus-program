#include<stdio.h>

int main(void)
{
    FILE* cfPtr; 
       
    if ((cfPtr = fopen("clients.txt", "w")) == NULL) { 
        puts("File could not be opened"); 
    }
    else { 
        puts("Enter the id, name, computer and math.");
        puts("Enter EOF to end input."); 
        printf("%s", "? ");

        unsigned int id; 
        char name[30];
        int computer, math;
        double avg=1.0;

        scanf("%d%29s%3d%3d", &id, name, &computer,&math);

        
               
        while (!feof(stdin)) { 
        	avg = ((float)computer + math) / 2;
            fprintf(cfPtr, "%d %s %d %d %.2f\n", id, name, computer, math, avg);            
            printf("%s", "? ");
            scanf("%d%29s%3d%3d", &id, name, &computer, &math);
        }

        fclose(cfPtr);    
    }
}
