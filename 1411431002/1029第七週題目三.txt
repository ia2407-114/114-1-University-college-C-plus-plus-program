#include<stdio.h>
#include<math.h>

float square(float x); //x為傳入的項目總數

int main(void) {

	//題目3: 請根據第六週的題目，寫一個副程式，傳入計算的項目總數n，計算下列級數的結果去近似pi值，並且傳回近似值為浮點數，最後由主程式印出結果


	printf("\n\n題目4\n\n");

	int n;
	float result;

	printf("請輸入項目總數n:");
	scanf_s("%d", &n);

	result = square(n);

	
	printf("項目數量為%d,結果為%.8f(取到小數點後八位)\n", n, result);

	return 0;
}

float square(float x) {

	int coun = 2, a1 = 1; //coun為計數器,a1為分母計數器(等差用)
	float pi = 4.0, answ1, a = -1.0; //pi為結果,a為分母

	for (coun = 2;coun <= x;coun++) {
		a = coun + a1;  //分母
		answ1 = 4 * (1 / a);
		if (coun % 2 == 0) {
			pi = pi + answ1 * -1;
		}
		else {
			pi = pi + answ1;
		}
		a1++; // a1 = a1 + 1
	}

	return pi;

}
