double app_pi(double x) {

	int coun = 2, a1 = 1;
	float pi = 4.0, answ1, a = -1.0;

	for (coun = 2;coun <= x;coun++) {
		a = coun + a1; 
		answ1 = 4 * (1 / a);
		if (coun % 2 == 0) {
			pi = pi + answ1 * -1;
		}
		else {
			pi = pi + answ1;
		}
		a1++;
	}

	return pi;
}

double ball_vol(double pi1, double r1) {
	return (4.0 / 3) * pi1 * r1 * r1 * r1;
}

double ball_area(double pi2, double r2) {
	return 4 * pi2 * r2 * r2;
}
double cyl_vol(double pi3, double r3, double h) {
	return (pi3 * r3 * r3) - h;
}
double cyl_area(double pi4, double r4, double h2) {
	return (2 * pi4 * r4 * r4) + (2 * pi4 * r4 * h2);
}
