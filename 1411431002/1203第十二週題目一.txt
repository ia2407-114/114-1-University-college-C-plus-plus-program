int ans (int n);
int main() {
    int n, i ;
    double e=1.0 ;
    printf("請輸入n:");
    scanf("%d/n",&n);
    for (i=1;i<=n;i++){
        e=e+ans(i);
    }
    printf("e=%.5f",e);
    return 0;
}
int ans(int x){
    if (x == 0){
        return 1;
    }
    else{
        return x*ans(x-1);
    }
}
