double ballv(double r, double pi) {
	
	return  4.0 / 3 * pi * r * r * r;
}
double pi(double k, double h, double i, double j, double pi2, double pi, double n) {
	for (;k != n;k++) {
		pi += 4 / i;
		i += 4;
	}
	for (;h != n;h++) {
		pi2 += 4 / j;
		j += 4;
	}
	return pi -= pi2;
}
double ballc(double r, double pi) {
	return 4.0 * pi * r * r;
}
double ballb(double r, double pi, double h) {
	return pi * r * r * h;
}
double balla(double r, double pi, double h) {
	return 2 * pi * r*(h + r);
}
